SELECT
    r.registration_id AS `Registration ID`,
    FROM_UNIXTIME(r.created) AS `Date created`,
    fn.field_full_name_value AS `Full Name`,
    r.anon_mail AS `E-mail`,
    GROUP_CONCAT(DISTINCT org.field_sb_organization_value ORDER BY org.delta SEPARATOR ', ') AS `Organization`,
    GROUP_CONCAT(DISTINCT sess.field_proffered_session_value ORDER BY sess.delta SEPARATOR ', ') AS `Preferred Session`,
    GROUP_CONCAT(DISTINCT pos.field_positions_value ORDER BY pos.delta SEPARATOR ', ') AS `Positions`,
    COALESCE(
        GROUP_CONCAT(DISTINCT stype.field_shirt_type_value ORDER BY stype.delta SEPARATOR ', '),
        'N/A'
    ) AS `Shirt Type`,
    GROUP_CONCAT(DISTINCT ssize.field_shirt_size_value ORDER BY ssize.delta SEPARATOR ', ') AS `Shirt Size`,
    GROUP_CONCAT(DISTINCT falg.field_food_alg_value ORDER BY falg.delta SEPARATOR ', ') AS `Food Allergies?`
FROM node n
LEFT JOIN registration r
    ON r.entity_id = n.nid
    AND r.entity_type = 'node'
LEFT JOIN field_data_field_full_name fn
    ON fn.entity_id = r.registration_id
    AND fn.entity_type = 'registration'
    AND fn.deleted = 0
LEFT JOIN field_data_field_sb_organization org
    ON org.entity_id = r.registration_id
    AND org.entity_type = 'registration'
    AND org.deleted = 0
LEFT JOIN field_data_field_proffered_session sess
    ON sess.entity_id = r.registration_id
    AND sess.entity_type = 'registration'
    AND sess.deleted = 0
LEFT JOIN field_data_field_positions pos
    ON pos.entity_id = r.registration_id
    AND pos.entity_type = 'registration'
    AND pos.deleted = 0
LEFT JOIN field_data_field_shirt_type stype
    ON stype.entity_id = r.registration_id
    AND stype.entity_type = 'registration'
    AND stype.deleted = 0
LEFT JOIN field_data_field_shirt_size ssize
    ON ssize.entity_id = r.registration_id
    AND ssize.entity_type = 'registration'
    AND ssize.deleted = 0
LEFT JOIN field_data_field_food_alg falg
    ON falg.entity_id = r.registration_id
    AND falg.entity_type = 'registration'
    AND falg.deleted = 0
WHERE n.status = 1
  AND n.type = 'form_pages'
  AND r.type = 'science_bowl_volunteer'
GROUP BY
    r.registration_id,
    r.created,
    fn.field_full_name_value,
    r.anon_mail,
    n.created
ORDER BY n.created DESC, r.created DESC;
