case 'image':
            $images = is_array($val) ? $val : [trim((string)$val)];
            $images = array_values(array_filter(array_map(static fn($u) => trim((string)$u), $images)));
            if (empty($images)) {
                return '';
            }
            $out = [];
            foreach ($images as $img_url) {
                $out[] = '<img src="' . htmlspecialchars($img_url) . '" alt="" style="max-width: 100%; height: auto; border-radius: 4px; border: 1px solid #e2e8f0; margin-top: 5px;">';
            }
            return count($out) > 1
                ? '<div class="field-image-gallery" style="display:flex; flex-wrap:wrap; gap:12px;">' . implode('', $out) . '</div>'
                : $out[0];

        case 'video':
            $videos = is_array($val) ? $val : [trim((string)$val)];
            $videos = array_values(array_filter(array_map(static fn($u) => trim((string)$u), $videos)));
            if (empty($videos)) {
                return '';
            }
            $embeds = [];
            foreach ($videos as $vid_url) {
                $vid_id = get_youtube_id($vid_url);
                if ($vid_id !== null) {
                    $embeds[] = '<div class="field-video-embed" style="position:relative; padding-bottom:56.25%; height:0; overflow:hidden; max-width:100%; border-radius:6px; border:1px solid #e2e8f0; margin-top:6px;">'
                              . '<iframe src="https://www.youtube-nocookie.com/embed/' . htmlspecialchars($vid_id) . '" title="YouTube video player" style="position:absolute; top:0; left:0; width:100%; height:100%; border:0;" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" referrerpolicy="strict-origin-when-cross-origin" allowfullscreen loading="lazy"></iframe>'
                              . '</div>';
                } else {
                    $embeds[] = '<a href="' . htmlspecialchars($vid_url) . '" target="_blank" rel="noopener">' . htmlspecialchars($vid_url) . '</a>';
                }
            }
            return implode('', $embeds);
